4. FILE: tests/test_task_manager.py

import tempfile import unittest from pathlib import Path from
unittest.mock import patch

import task_manager

class TaskManagerTests(unittest.TestCase):

    def setUp(self):
        self.temp_dir = tempfile.TemporaryDirectory()
        self.data_file = Path(self.temp_dir.name) / "tasks.json"
        self.patcher = patch.object(task_manager, "DATA_FILE", self.data_file)
        self.patcher.start()

    def tearDown(self):
        self.patcher.stop()
        self.temp_dir.cleanup()

    def test_add_task(self):
        task = task_manager.add_task("Prepare assignment")
        self.assertEqual(task["id"], 1)
        self.assertEqual(task["title"], "Prepare assignment")
        self.assertFalse(task["completed"])

    def test_empty_title_is_rejected(self):
        with self.assertRaises(ValueError):
            task_manager.add_task("   ")

    def test_complete_task(self):
        task = task_manager.add_task("Learn Git")
        self.assertTrue(task_manager.complete_task(task["id"]))
        self.assertTrue(task_manager.list_tasks()[0]["completed"])

    def test_delete_task(self):
        task = task_manager.add_task("Push code to GitHub")
        self.assertTrue(task_manager.delete_task(task["id"]))
        self.assertEqual(task_manager.list_tasks(), [])

    def test_unknown_task_id(self):
        self.assertFalse(task_manager.complete_task(999))
        self.assertFalse(task_manager.delete_task(999))

A simple command-line task management application built with Python.
Users can add tasks, view saved tasks, mark tasks as completed, and
